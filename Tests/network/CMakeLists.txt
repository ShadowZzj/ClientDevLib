cmake_minimum_required(VERSION 3.20)
project(ClientDevLibNetworkTests LANGUAGES CXX)
include(CTest)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ClientNetwork.cmake")
add_executable(client_network_tests NetworkTests.cpp)
target_link_libraries(client_network_tests PRIVATE ClientDevLib::Network)
if(MSVC)
    target_compile_options(client_network PRIVATE /W4 /WX)
    target_compile_options(client_network_tests PRIVATE /W4 /WX)
endif()
add_test(NAME client_network_parser COMMAND client_network_tests)
if(WIN32)
    add_executable(client_windows_network_tests WindowsNetworkTests.cpp)
    target_link_libraries(client_windows_network_tests PRIVATE ClientDevLib::WindowsNetwork)
    if(MSVC)
        target_compile_options(client_windows_network PRIVATE /W4 /WX)
        target_compile_options(client_windows_network_tests PRIVATE /W4 /WX)
    endif()
    add_test(NAME client_windows_network_helpers COMMAND client_windows_network_tests)
endif()
